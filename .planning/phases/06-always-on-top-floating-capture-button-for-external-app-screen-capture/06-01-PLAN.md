Create the floating overlay window, its preload IPC bridge, and the overlay React mini-app with idle/recording/finalizing visual states. Also update Vite to build the overlay as a second HTML entry point.

1. Create `apps/desktop/src/main/preload-capture.js` using CommonJS (matches existing preload.js pattern):
   ```javascript
   const { contextBridge, ipcRenderer } = require('electron');

   contextBridge.exposeInMainWorld('captureApi', {
     startSession: () => ipcRenderer.invoke('capture:start-session'),
     stopSession: () => ipcRenderer.invoke('capture:stop-session'),
     getSessionState: () => ipcRenderer.invoke('capture:get-state'),
     onStateChange: (callback) => {
       const handler = (_event, state) => callback(state);
       ipcRenderer.on('capture:state-changed', handler);
       return () => ipcRenderer.removeListener('capture:state-changed', handler);
     },
   });
   ```
   The bridge exposes exactly 4 methods on `window.captureApi`. No backend client, no file picker -- this is a minimal overlay preload. The `onStateChange` returns a cleanup function.

2. Create `apps/desktop/src/renderer/capture-overlay/CaptureOverlay.jsx`:

   ```jsx
   import React, { useCallback, useEffect, useState } from 'react';
   import './capture-overlay.css';

   // Inline SVG icons (no external icon library in overlay -- per UI-SPEC)
   function CameraIcon() {
     return (
       <svg viewBox="0 0 24 24">
         <path d="M23 19a2 2 0 0 1-2 2H3a2 2 0 0 1-2-2V8a2 2 0 0 1 2-2h4l2-3h6l2 3h4a2 2 0 0 1 2 2z" />
         <circle cx="12" cy="13" r="4" />
       </svg>
     );
   }

   function StopIcon() {
     return (
       <svg viewBox="0 0 24 24">
         <rect x="6" y="6" width="12" height="12" rx="1" fill="currentColor" stroke="none" />
       </svg>
     );
   }

   function SpinnerIcon() {
     return (
       <svg className="capture-spinner" viewBox="0 0 24 24">
         <path d="M12 2v4M12 18v4M4.93 4.93l2.83 2.83M16.24 16.24l2.83 2.83M2 12h4M18 12h4M4.93 19.07l2.83-2.83M16.24 7.76l2.83-2.83" />
       </svg>
     );
   }

   export default function CaptureOverlay() {
     const [state, setState] = useState('idle'); // idle | recording | finalizing

     useEffect(() => {
       if (!window.captureApi) return;

       // Sync initial state
       window.captureApi.getSessionState().then((s) => {
         if (s) setState(s);
       });

       // Listen for state changes from main process
       const cleanup = window.captureApi.onStateChange((newState) => {
         setState(newState);
       });

       return cleanup;
     }, []);

     const handleClick = useCallback(async () => {
       if (!window.captureApi) return;
       if (state === 'finalizing') return; // Ignore clicks while finalizing

       if (state === 'idle') {
         await window.captureApi.startSession();
       } else if (state === 'recording') {
         await window.captureApi.stopSession();
       }
     }, [state]);

     const buttonClass = `capture-button capture-button--${state}`;

     const icon =
       state === 'idle' ? <CameraIcon /> :
       state === 'recording' ? <StopIcon /> :
       <SpinnerIcon />;

     const tooltip = state === 'idle' ? 'Start screen capture' : state === 'recording' ? 'Stop screen capture' : 'Creating note...';

     return (
       <div className="capture-overlay-root">
         <div className="capture-button-container">
           {state === 'recording' && <div className="capture-pulse-ring" />}
           <button
             className={buttonClass}
             onClick={handleClick}
             title={tooltip}
             aria-label={tooltip}
           >
             {icon}
           </button>
         </div>
       </div>
     );
   }
   ```

   The component has 3 visual states per UI-SPEC:
   - idle: dark primary fill, camera icon, hover scale effect
   - recording: destructive red fill, stop square icon, pulsing ring animation
   - finalizing: muted fill, spinning loader icon, no pulse, clicks ignored

   State is driven entirely by the main process via `captureApi.onStateChange`. The overlay does NOT own state -- it reflects what the main process tells it.
