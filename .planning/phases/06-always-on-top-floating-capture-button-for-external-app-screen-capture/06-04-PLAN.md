<objective>
Add macOS Accessibility tree text extraction as the primary text extraction method, with existing OCR as fallback. Build a standalone Swift CLI binary that walks the AX tree of a given PID and outputs JSON text, plus a Node.js wrapper to invoke it.

Purpose: Per the locked decision in CONTEXT.md, macOS Accessibility API is the primary text extraction strategy. This produces cleaner, more structured text than OCR for apps that expose their accessibility tree (Slack, Chrome, Safari, Terminal, etc.). OCR remains as the fallback for apps without good accessibility support.

Output: A compiled Swift binary `ax_walker` invoked via child_process, a Node.js wrapper `ax-tree-extractor.js`, and updated screen-capture.js that tries AX tree first and falls back to OCR.
</objective>

1. Create `apps/desktop/src/main/native/ax_walker.swift` -- a standalone Swift CLI program that:
   - Accepts a PID as the first command-line argument
   - Creates an AXUIElement for that PID via `AXUIElementCreateApplication(pid)`
   - Recursively walks the element tree collecting all elements with `AXValue` or `AXTitle` or `AXDescription` attributes that contain text
   - Filters out empty strings, window chrome elements (close/minimize/zoom buttons), and menu bar items
   - Outputs a JSON object to stdout: `{"texts": ["line1", "line2", ...], "elementCount": N, "success": true}`
   - If the AX tree is inaccessible or empty, outputs: `{"texts": [], "elementCount": 0, "success": false, "error": "reason"}`
   - Exits with code 0 on success, 1 on error

   ```swift
   import Foundation
   import ApplicationServices

   struct AXResult: Codable {
       let texts: [String]
       let elementCount: Int
       let success: Bool
       let error: String?
   }

   func extractTexts(from element: AXUIElement, depth: Int = 0, maxDepth: Int = 15) -> [String] {
       guard depth < maxDepth else { return [] }

       var texts: [String] = []

       // Get the role to skip non-content elements
       var roleRef: CFTypeRef?
       AXUIElementCopyAttributeValue(element, kAXRoleAttribute as CFString, &roleRef)
       let role = roleRef as? String ?? ""

       // Skip menu bars, menu items, and window control buttons
       let skipRoles: Set<String> = ["AXMenuBar", "AXMenu", "AXMenuItem", "AXMenuBarItem"]
       if skipRoles.contains(role) { return [] }

       // Extract text value
       var valueRef: CFTypeRef?
       AXUIElementCopyAttributeValue(element, kAXValueAttribute as CFString, &valueRef)
       if let value = valueRef as? String, !value.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
           texts.append(value.trimmingCharacters(in: .whitespacesAndNewlines))
       }

       // Extract title (for buttons, labels, etc.)
       var titleRef: CFTypeRef?
       AXUIElementCopyAttributeValue(element, kAXTitleAttribute as CFString, &titleRef)
       if let title = titleRef as? String, !title.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
           // Skip generic window chrome titles
           let skipTitles: Set<String> = ["Close", "Minimize", "Zoom", "Full Screen"]
           if !skipTitles.contains(title) {
               texts.append(title)
           }
       }

       // Extract description
       var descRef: CFTypeRef?
       AXUIElementCopyAttributeValue(element, kAXDescriptionAttribute as CFString, &descRef)
       if let desc = descRef as? String, !desc.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
           texts.append(desc)
       }

       // Recurse into children
       var childrenRef: CFTypeRef?
       AXUIElementCopyAttributeValue(element, kAXChildrenAttribute as CFString, &childrenRef)
       if let children = childrenRef as? [AXUIElement] {
           for child in children {
               texts.append(contentsOf: extractTexts(from: child, depth: depth + 1, maxDepth: maxDepth))
           }
       }

       return texts
   }

   // Main
   guard CommandLine.arguments.count >= 2,
         let pid = pid_t(CommandLine.arguments[1]) else {
       let result = AXResult(texts: [], elementCount: 0, success: false, error: "Usage: ax_walker <pid>")
       let data = try! JSONEncoder().encode(result)
       FileHandle.standardOutput.write(data)
       exit(1)
   }

   let appElement = AXUIElementCreateApplication(pid)
   let texts = extractTexts(from: appElement)

   // Deduplicate while preserving order
   var seen = Set<String>()
   let uniqueTexts = texts.filter { text in
       guard !seen.contains(text) else { return false }
       seen.insert(text)
       return true
   }

   let result = AXResult(
       texts: uniqueTexts,
       elementCount: uniqueTexts.count,
       success: !uniqueTexts.isEmpty
   )

   let data = try! JSONEncoder().encode(result)
   FileHandle.standardOutput.write(data)
   FileHandle.standardOutput.write("\n".data(using: .utf8)!)
   exit(0)
   ```

5. Create `apps/desktop/src/main/ax-tree-extractor.js` (CommonJS):
   ```javascript
   const { execFile } = require('child_process');
   const path = require('path');

   const AX_WALKER_PATH = path.join(__dirname, 'native', 'ax_walker');
   const TIMEOUT_MS = 3000;

   function extractTextFromAccessibilityTree(pid) {
     return new Promise((resolve) => {
       if (!pid || pid <= 0) {
         resolve({ texts: [], success: false, method: 'ax-skipped', error: 'No PID' });
         return;
       }

       execFile(AX_WALKER_PATH, [String(pid)], { timeout: TIMEOUT_MS }, (error, stdout) => {
         if (error) {
           resolve({ texts: [], success: false, method: 'ax-failed', error: error.message });
           return;
         }

         try {
           const result = JSON.parse(stdout.trim());
           resolve({
             texts: result.texts || [],
             success: result.success === true,
             method: result.success ? 'accessibility' : 'ax-empty',
             error: result.error || null,
           });
         } catch (parseError) {
           resolve({ texts: [], success: false, method: 'ax-parse-error', error: parseError.message });
         }
       });
     });
   }

   module.exports = { extractTextFromAccessibilityTree };
   ```

   Key design decisions:
   - Uses `child_process.execFile` (not exec) for safety -- no shell injection
   - 3-second timeout prevents hanging if the AX tree is enormous
   - Never throws -- always resolves with a result object (success: true/false)
   - Returns `method` field so the capture session knows which extractor succeeded
   - The Swift binary path is resolved relative to `__dirname` (works in dev and packaged)
