**BROWSER_BUNDLE_IDS** -- a Set containing `'com.google.Chrome'` and `'com.apple.Safari'` (per D-06).

**getBrowserTabUrl(bundleId)** -- async function that uses `child_process.execFile('osascript', ...)` to get the current tab URL from Chrome or Safari. Per D-04, the user said "AX tree URL extraction" as primary, but research (08-RESEARCH.md Open Question 1) shows the existing ax_walker.m returns all text content, not URLs specifically. AppleScript accesses the same scripting bridge more reliably. Using AppleScript as the implementation of URL extraction per user decision D-04 (research recommended this approach). Falls back to empty string on error (per Pitfall 1 -- permission prompts may occur on first run). Set timeout to 1000ms.

For Safari: `tell app "Safari" to get URL of current tab of front window`
For Chrome: `tell app "Google Chrome" to get URL of active tab of front window`

Important: Only call getBrowserTabUrl when active-win already reports a browser in the foreground. Never call when the browser is not frontmost (per Pitfall 2 -- AppleScript would launch the browser).

**shouldShowOverlay(windowInfo, whitelist)** -- async function that determines whether the overlay should be visible:
1. If no apps are enabled in whitelist, return false immediately
2. Check each enabled app in the whitelist:
   a. If app has a `match` function (native app detection), test against windowInfo. Return true on first match.
   b. If active window is a browser (bundleId in BROWSER_BUNDLE_IDS), call getBrowserTabUrl(bundleId) and test each enabled app's `matchUrl`. If URL extraction returns empty, fall back to testing each app's `matchTitle` against windowInfo.windowTitle (per D-05).
   c. If active window is NOT a browser, still check `matchTitle` for apps that have it (e.g., Notion desktop is separate from Notion in browser, but title fallback still applies).
3. Return false if no match found.

3. Add whitelist persistence functions (main-process-owned JSON file, NOT backend DB -- per RESEARCH.md recommendation, this is desktop-only UI state):
```javascript
function getWhitelistPath() {
  return path.join(app.getPath('userData'), 'capture-apps.json');
}

function loadWhitelist() {
  try {
    const data = JSON.parse(require('fs').readFileSync(getWhitelistPath(), 'utf8'));
    // Only return keys that are valid preset app IDs
    const result = {};
    for (const key of Object.keys(PRESET_APPS)) {
      result[key] = Boolean(data[key]);
    }
    return result;
  } catch {
    return {}; // Empty by default (D-01)
  }
}

function saveWhitelist(whitelist) {
  require('fs').writeFileSync(getWhitelistPath(), JSON.stringify(whitelist, null, 2));
}
```

4. Add polling functions:
```javascript
function startAppDetection() {
  if (appDetectionTimer) return;
  const { getActiveWindowInfo } = require('./screen-capture');

  appDetectionTimer = setInterval(async () => {
    // Always show overlay during active capture session (Pitfall 4)
    if (captureSession && captureSession.state === 'capturing') {
      if (captureOverlay && !captureOverlay.isDestroyed() && !captureOverlay.isVisible()) {
        captureOverlay.show();
      }
      return;
    }

    try {
      const windowInfo = await getActiveWindowInfo();
      const whitelist = loadWhitelist();
      const shouldShow = await shouldShowOverlay(windowInfo, whitelist);

      if (shouldShow && captureOverlay && !captureOverlay.isDestroyed() && !captureOverlay.isVisible()) {
        captureOverlay.show();
      } else if (!shouldShow && captureOverlay && !captureOverlay.isDestroyed() && captureOverlay.isVisible()) {
        captureOverlay.hide();
      }
    } catch {
      // Silently skip failed detection cycles
    }
  }, 500); // 500ms polling interval per RESEARCH.md recommendation
}

function stopAppDetection() {
  if (appDetectionTimer) {
    clearInterval(appDetectionTimer);
    appDetectionTimer = null;
  }
}
```

5. Add IPC handlers inside `registerIpcHandlers()`:
```javascript
ipcMain.handle('capture-apps:get-presets', () => {
  return Object.values(PRESET_APPS).map(app => ({
    id: app.id,
    label: app.label,
    description: app.description,
  }));
});

ipcMain.handle('capture-apps:get', () => loadWhitelist());

ipcMain.handle('capture-apps:update', (_event, whitelist) => {
  saveWhitelist(whitelist);
  const hasEnabled = Object.values(whitelist).some(Boolean);
  if (hasEnabled) {
    startAppDetection();
  } else {
    stopAppDetection();
    if (captureOverlay && !captureOverlay.isDestroyed() && captureOverlay.isVisible()) {
      captureOverlay.hide();
    }
  }
  return whitelist;
});
```
