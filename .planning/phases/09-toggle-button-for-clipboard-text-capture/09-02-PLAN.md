```javascript
const clipBtn = document.getElementById('clipBtn');
const clipIcon = document.getElementById('clipIcon');
const badge = document.getElementById('badge');

const CLIP_ICONS = {
  idle: `<path d="M16 4h2a2 2 0 0 1 2 2v14a2 2 0 0 1-2 2H6a2 2 0 0 1-2-2V6a2 2 0 0 1 2-2h2"/><rect x="8" y="2" width="8" height="4" rx="1" ry="1"/>`,
  monitoring: `<path d="M16 4h2a2 2 0 0 1 2 2v14a2 2 0 0 1-2 2H6a2 2 0 0 1-2-2V6a2 2 0 0 1 2-2h2"/><rect x="8" y="2" width="8" height="4" rx="1" ry="1"/><path d="M9 14l2 2 4-4"/>`,
  finalizing: `<path class="spinner" d="M12 2v4M12 18v4M4.93 4.93l2.83 2.83M16.24 16.24l2.83 2.83M2 12h4M18 12h4M4.93 19.07l2.83-2.83M16.24 7.76l2.83-2.83"/>`,
};

const CLIP_TITLES = {
  idle: 'Start clipboard monitoring',
  monitoring: 'Stop clipboard monitoring',
  finalizing: 'Creating note...',
};

function setClipState(state) {
  const cssClass = state === 'monitoring' ? 'monitoring' : (state === 'finalizing' ? 'clip-finalizing' : '');
  clipBtn.className = 'btn' + (cssClass ? ' ' + cssClass : '');
  clipIcon.innerHTML = CLIP_ICONS[state] || CLIP_ICONS.idle;
  clipBtn.title = CLIP_TITLES[state] || CLIP_TITLES.idle;
  clipBtn.setAttribute('aria-label', clipBtn.title);
  if (state === 'idle') {
    badge.style.display = 'none';
    badge.textContent = '';
  }
}

function updateBadge(count) {
  if (count > 0) {
    badge.style.display = 'block';
    badge.textContent = count > 99 ? '99+' : String(count);
  } else {
    badge.style.display = 'none';
    badge.textContent = '';
  }
}

if (window.captureApi) {
  window.captureApi.getClipboardState().then(s => { if (s) setClipState(s); });
  window.captureApi.onClipboardStateChange(setClipState);
  window.captureApi.onClipboardCount(updateBadge);
}

clipBtn.addEventListener('click', async () => {
  if (!window.captureApi) {
    console.warn('[capture-overlay] captureApi not available');
    return;
  }
  if (clipBtn.classList.contains('clip-finalizing')) return;

  if (clipBtn.classList.contains('monitoring')) {
    await window.captureApi.stopClipboard();
  } else {
    await window.captureApi.startClipboard();
  }
});
```
