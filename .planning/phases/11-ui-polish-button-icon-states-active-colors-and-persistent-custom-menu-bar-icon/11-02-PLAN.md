Create the `apps/desktop/resources/` directory and generate four PNG files for the macOS tray icon.

Write a Node.js script (temporary, run once, then delete) that generates the PNGs programmatically. Use the `canvas` npm package (install as devDependency: `npm install --save-dev canvas --workspace @privanote/desktop`) to draw:

**trayTemplate.png (16x16) and trayTemplate@2x.png (32x32):**
- Background: fully transparent (alpha = 0)
- Glyph: the letter "P" in a bold sans-serif font (e.g., "Helvetica Neue Bold" or "Arial Bold")
- Color: solid black (#000000, full opacity) — macOS template image convention; macOS will invert for dark mode
- The P should be centered in the canvas, filling roughly 12x12 of the 16x16 space (24x24 of 32x32 for @2x)
- Font size: ~14px for 16x16, ~28px for 32x32
- Use `ctx.fillText()` centered with `textAlign: 'center'` and `textBaseline: 'middle'`

**trayRecTemplate.png (16x16) and trayRecTemplate@2x.png (32x32):**
- Same P glyph as above
- Add a small filled red circle (#FF3B30, Apple's system red) at the bottom-right corner:
  - 16x16: 4px diameter circle centered at (13, 13)
  - 32x32: 8px diameter circle centered at (26, 26)
- The red dot is NOT a template color (it stays red regardless of dark/light mode). This means the recording variant should NOT use `setTemplateImage(true)` — instead it will be a regular image. The P glyph in this variant should use white (#FFFFFF) for dark menu bars and black (#000000) for light menu bars. Since we cannot detect menu bar theme from main process easily, use black P + red dot and do NOT set as template. macOS will show it as-is. Alternatively, to keep the template behavior for the P, create the PNG with black P + red dot and set as template — macOS will invert the black P but leave the red dot as-is since template images only invert grayscale pixels. Red (non-grayscale) pixels are preserved. This is the correct approach.

So all four PNGs use black P on transparent background, and the recording variants add a red (#FF3B30) dot. All four can be set as template images — macOS preserves colored pixels and only inverts grayscale.

Script outline:
```javascript
const { createCanvas } = require('canvas');
const fs = require('fs');
const path = require('path');

function drawIcon(size, withDot) {
  const canvas = createCanvas(size, size);
  const ctx = canvas.getContext('2d');
  // Transparent background (default)
  ctx.clearRect(0, 0, size, size);
  // Draw P
  ctx.fillStyle = '#000000';
  ctx.font = `bold ${Math.round(size * 0.875)}px "Helvetica Neue"`;
  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';
  ctx.fillText('P', size / 2, size / 2);
  // Red dot for recording variant
  if (withDot) {
    const dotR = size * 0.125; // 2px radius at 16, 4px at 32
    ctx.fillStyle = '#FF3B30';
    ctx.beginPath();
    ctx.arc(size - dotR - 1, size - dotR - 1, dotR, 0, Math.PI * 2);
    ctx.fill();
  }
  return canvas.toBuffer('image/png');
}

const outDir = path.join(__dirname, 'apps', 'desktop', 'resources');
fs.mkdirSync(outDir, { recursive: true });
fs.writeFileSync(path.join(outDir, 'trayTemplate.png'), drawIcon(16, false));
fs.writeFileSync(path.join(outDir, 'trayTemplate@2x.png'), drawIcon(32, false));
fs.writeFileSync(path.join(outDir, 'trayRecTemplate.png'), drawIcon(16, true));
fs.writeFileSync(path.join(outDir, 'trayRecTemplate@2x.png'), drawIcon(32, true));
console.log('Tray icons generated');
```

```javascript
function updateTray(state) {
  if (!tray) return;

  const isRecording = state === 'capturing' || state === 'recording';

  // Swap tray icon between idle P and recording P+red-dot (per D-12, D-14)
  tray.setImage(createTrayIcon(isRecording));

  if (isRecording) {
    // Remove emoji title — icon alone signals state (per D-12)
    tray.setTitle('');
    tray.setToolTip('Privanote -- Recording...');
    tray.setContextMenu(Menu.buildFromTemplate([
      { label: 'Recording...', enabled: false },
      { label: 'Stop Capture', click: () => toggleCaptureSession() },
      { type: 'separator' },
      { label: 'Show Privanote', click: () => { if (mainWindow) { mainWindow.show(); mainWindow.focus(); } } },
      { label: 'Quit Privanote', click: () => { app.quit(); } },
    ]));
  } else {
    tray.setTitle('');
    tray.setToolTip('Privanote Capture');
    tray.setContextMenu(Menu.buildFromTemplate([
      { label: 'Start Capture', click: () => toggleCaptureSession() },
      { type: 'separator' },
      { label: 'Show Privanote', click: () => { if (mainWindow) { mainWindow.show(); mainWindow.focus(); } } },
      { label: 'Quit Privanote', click: () => { app.quit(); } },
    ]));
  }

  // Dock badge — shows "REC" on the app icon in the taskbar when recording
  if (process.platform === 'darwin' && app.dock) {
    app.dock.setBadge(isRecording ? 'REC' : '');
  }
}
```
